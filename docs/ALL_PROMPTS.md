# All Prompts, Combined

Every LLM prompt used by the simulator, in one place. Generated by [`scripts/build_all_prompts.py`](../scripts/build_all_prompts.py) directly from the source, so the text below is verbatim. Prompts are Python f-strings: `{braces}` are filled in at runtime. Where a prompt is assembled from many fragments, a **rendered example** follows the source.

All text generation uses `gemini-2.5-flash`; embeddings use `gemini-embedding-001` (no prompt). Prompts marked *(no LLM)* describe deterministic logic included for context.

```python
def _build_clickstream(self, db: Session, agent_id: str, session_id: str, current_tick: int) -> str:
    min_tick = max(0, current_tick - 10)
    events = (
        db.query(Event)
        .filter(
            Event.agent_id == agent_id,
            Event.session_id == session_id,
            Event.sim_tick >= min_tick,
        )
        .order_by(Event.sim_tick.asc())
        .limit(50)
        .all()
    )

    if not events:
        return "No recent activity."

    lines = []
    for e in events:
        snippet = ""
        if e.target_id:
            post = db.query(DeliberationPost).filter(DeliberationPost.id == e.target_id).first()
            if post:
                snippet = f'on "{post.name[:50]}" ({post.ptype})'
        meta_parts = []
        if e.metadata_:
            if "ptype" in e.metadata_:
                meta_parts.append(f"ptype: {e.metadata_['ptype']}")
            if "new_post_name" in e.metadata_:
                meta_parts.append(f"added: {e.metadata_['new_post_name'][:40]}")
            if "reason" in e.metadata_:
                meta_parts.append(f"reason: {e.metadata_['reason']}")
        meta_str = f" — {', '.join(meta_parts)}" if meta_parts else ""
        lines.append(f"[tick {e.sim_tick}] {e.event_type} {snippet}{meta_str}")

    return "\n".join(lines)
```

```python
def _build_selector_prompt(
    self,
    profile: str,
    valid_pairs: list,
    persona_topics: list[str] | None = None,
    skip_reasons: list[str] | None = None,
    learned: dict | None = None,
) -> str:
    pair_lines = [
        f"[{i}] Action: {p['action']} | On: ({p['post']['ptype']}) {p['post']['name'][:60]}"
        for i, p in enumerate(valid_pairs[:20])
    ]
    persona_block = ""
    if persona_topics:
        persona_block = (
            "PERSONA TABLE (topics this expert said they want to contribute to):\n"
            + ", ".join(persona_topics) + "\n\n"
        )
    # Quantitative feedback signal — how often the expert has engaged with each
    # recommended action so far this session. This is what lets the recommender
    # improve (lower its skip rate) as the run progresses.
    learned_block = ""
    if learned and learned.get("n", 0) > 0:
        rate_lines = sorted(learned.get("rates", {}).items(), key=lambda kv: kv[1], reverse=True)
        learned_block = (
            "ENGAGEMENT RATES SO FAR (this expert, this session — higher = they act, "
            "lower = they skip):\n"
            + "\n".join(f"- {a}: {r:.0%}" for a, r in rate_lines)
            + f"\n(overall {learned['base']:.0%} over {learned['n']} decisions). "
            "Strongly prefer higher-rate actions.\n\n"
        )
    skip_block = ""
    if skip_reasons:
        skip_block = (
            "\n\nRECENT SKIP REASONS (avoid recommending pairs like these):\n"
            + "\n".join(f"- {s}" for s in skip_reasons)
        )
    return (
        persona_block
        + learned_block
        + f"BEHAVIORAL PROFILE (learned from clickstream):\n{profile}\n\n"
        + f"AVAILABLE POST/ACTION PAIRS:\n" + "\n".join(pair_lines)
        + skip_block + "\n\n"
        "Which pair (by index) is this expert most likely to engage with?"
    )
```

`backend/simulation/recommender.py:341` — `RecommendationEngine._rank_pairs_by_learning`

```python
def _rank_pairs_by_learning(
    self, valid_pairs: list, learned: dict, all_posts: dict | None = None,
    fit_scores: dict | None = None, agent_id: str | None = None,
    tree_scores: dict | None = None,
    keep: int = 12, explore: int | None = None,
) -> list:
    """
    Rank candidate pairs by predicted engagement and keep the strongest, plus a
    little random exploration. Predicted engagement folds in signals that accumulate
    over a run, so the skip rate trends down as they sharpen:

      1. SEMANTIC FIT — similarity of the candidate post to what this agent has
         engaged with vs skipped (the dominant real driver of skips); falls back to
      2. learned per-action engagement rate until fit history exists,
      3. an OWN-FRONTIER boost — prefer extending the agent's own branches (posts it
         authored, or their direct children), which stay on-fit and unsaturated even
         as the wider tree drifts away from this agent's expertise,
      4. a penalty for posts the agent has already skipped (stop re-pushing them),
      5. a gap/saturation factor — free until TARGET_BREADTH siblings exist, then
         down-weight ("already covered" skips),
      6. depth and same-type-chain penalties, so the map grows OUT as well as DOWN
         instead of collapsing into one long drifting corridor.
    """
    from simulation.roles.specialist import ACTION_TO_PTYPE

    # Exploration schedule: sample MORE while the model is cold (broader coverage
    # trains the tree faster) and taper to pure exploitation once it has sharpened.
    # This is what steepens the early part of the skip-rate curve.
    if explore is None:
        n_seen = learned.get("n", 0)
        explore = (3 if n_seen < 16 else (2 if n_seen < 32 else (1 if n_seen < 60 else 0))) \
                  if FAST_LEARNING else 1

    rates, base = learned.get("rates", {}), learned.get("base", 0.5)
    skipped_posts = learned.get("skipped_posts", {})
    fit = fit_scores or {}
    tree = tree_scores or {}
    posts = all_posts or {}

    def _saturation(pair) -> float:
        """Free until TARGET_BREADTH siblings exist, then decay — so branches fan
        out to a healthy width before the recommender abandons them."""
        if not posts:
            return 1.0
        target_ptype = ACTION_TO_PTYPE.get(pair["action"])
        if not target_ptype:
            return 1.0
        existing = sum(
            1 for p in posts.values()
            if p.get("purpose_id") == pair["post"]["id"] and p.get("ptype") == target_ptype
        )
        if existing < TARGET_BREADTH:
            return 1.0
        return 1.0 / (1.0 + (existing - TARGET_BREADTH + 1))

    def _depth(post: dict) -> int:
        d, cur, seen = 0, post, set()
        while cur is not None:
            pid = cur.get("purpose_id")
            if not pid or pid in seen or pid not in posts:
                break
            seen.add(pid)
            cur = posts[pid]
            d += 1
        return d

    def _depth_penalty(pair) -> float:
        """Damp candidates far down the tree so the map grows out, not just down."""
        if not posts:
            return 1.0
        d = _depth(pair["post"])
        return 1.0 if d <= DEPTH_SOFT_CAP else DEPTH_DECAY ** (d - DEPTH_SOFT_CAP)

    def _chain_penalty(pair) -> float:
        """
        Damp long runs of the same post type — e.g. add_improvement on a solution
        produces another solution, which produced 7-deep solution chains that drifted
        away from the originating barrier.
        """
        if not posts:
            return 1.0
        target_ptype = ACTION_TO_PTYPE.get(pair["action"])
        if not target_ptype:
            return 1.0
        run, cur, seen = 0, pair["post"], set()
        while cur is not None and cur.get("ptype") == target_ptype:
            run += 1
            pid = cur.get("purpose_id")
            if not pid or pid in seen or pid not in posts:
                break
            seen.add(pid)
            cur = posts[pid]
        return 1.0 if run <= MAX_SAME_TYPE_CHAIN else CHAIN_DECAY ** (run - MAX_SAME_TYPE_CHAIN)

    def _own_frontier(pair) -> float:
        # Boost acting on the agent's own post, or a child of its own post — its
        # coherent, on-expertise lane where fresh, engageable openings keep appearing.
        if not agent_id:
            return 1.0
        post = pair["post"]
        if post.get("author_agent_id") == agent_id:
            return 1.6
        parent = posts.get(post.get("purpose_id")) if posts else None
        if parent and parent.get("author_agent_id") == agent_id:
            return 1.3
        return 1.0

    def _score(pair) -> float:
        pid = pair["post"]["id"]
        # Priority of predictors: trained decision tree > semantic fit > action-rate.
        if pid in tree:
            r = tree[pid]
        elif pid in fit:
            r = fit[pid]
        else:
            r = rates.get(pair["action"], base)
        r *= _own_frontier(pair)
        skips = skipped_posts.get(pid, 0)
        if skips:
            r *= 0.35 ** skips   # strongly avoid branches this agent keeps rejecting
        # Shape as a BOUNDED tie-breaker: combine the structural factors into one
        # 0..1 term and let it modulate only SHAPE_WEIGHT of the score, so a badly
        # shaped but well-matched candidate is demoted, never vetoed.
        shape = _saturation(pair) * _depth_penalty(pair) * _chain_penalty(pair)
        return r * (1.0 - SHAPE_WEIGHT + SHAPE_WEIGHT * shape)

    if len(valid_pairs) <= keep:
        return sorted(valid_pairs, key=_score, reverse=True)

    scored = sorted(valid_pairs, key=_score, reverse=True)
    top = scored[:keep]
    rest = scored[keep:]
    if rest and explore > 0:
        top = top + random.sample(rest, min(explore, len(rest)))
    return top
```

```python
def build_action_choice_prompt(
    agent_name: str,
    domain: str,
    post: dict,
    context_str: str,
    available_actions: list[str],
    recommended_action: str,
    existing_by_ptype: dict,
    all_posts: dict | None = None,
    use_aideator: bool = False,
    brief_answers: bool = True,
) -> str:
    """
    Agent sees a post and chooses which action to take from the available list.
    The recommended_action is shown as a suggestion (not forced).
    Agent responds with JSON including their chosen action and content.
    """

    # Build action menu
    action_lines = []
    for a in available_actions:
        marker = " ← SUGGESTED" if a == recommended_action else ""
        desc = ACTION_DESCRIPTIONS.get(a, a)
        action_lines.append(f'  "{a}" — {desc}{marker}')
    actions_str = "\n".join(action_lines)

    # Existing children summary (avoid duplicates)
    existing_parts = []
    for ptype, posts in existing_by_ptype.items():
        names = [p["name"] for p in posts[:5]]
        existing_parts.append(f"  {ptype}: {', '.join(names)}")
    existing_str = "\n".join(existing_parts) if existing_parts else "  (none yet)"

    brevity_block = (
        'BE BRIEF. The "description" must be 2-3 SENTENCES TOTAL — not per point, not per\n'
        "paragraph. Write it as one short paragraph, the way a real participant typing into a\n"
        "web form would. Ignore any guidance above that asks for multiple paragraphs or a\n"
        "paragraph per topic; keep the substance, drop the length.\n"
        "Do NOT open with a rhetorical question or a heading (no \"What is this?\", no\n"
        "\"Why is it important?\"). Do NOT use markdown or HTML. Start with the substance.\n\n"
    ) if brief_answers else ""

    # Content instruction for the suggested action (shown as guidance)
    suggestion_instruction = _content_instruction(
        recommended_action,
        post,
        all_posts if all_posts is not None else {},
        existing_by_ptype=existing_by_ptype,
        use_aideator=use_aideator,

    return f"""You are {agent_name}, a {domain} expert in a structured deliberation.

CONTEXT:
{context_str}

CURRENT POST [{post['ptype'].upper()}]: "{post['name']}"
{post.get('description', '')}

{existing_str}

{actions_str}

{suggestion_instruction}

Choose the action that best fits your expertise and the deliberation needs.
You may choose the suggested action or any other available action.

action. If you skip, the reason must explain what made this recommendation a poor

For content actions: {{"action": "add_barrier", "name": "...", "description": "...", "reason": "why this action"}}
```

```python
def describe_context(post: dict, all_posts: dict) -> str:
    """
    Walks from post up to root via purpose_id.
    Returns human-readable context chain. Mirrors Lisp describe-context.
    """
    chain = []
    current = post
    while current is not None:
        chain.append(current)
        pid = current.get("purpose_id")
        current = all_posts.get(pid) if pid else None
    chain.reverse()

    templates = {
        "mission":     "The mission is: {name}. {desc}",
        "stakeholder": "One stakeholder is: {name}. {desc}",
        "goal":        "A goal for this stakeholder is: {name}. {desc}",
        "barrier":     "A barrier is: {name}. {desc}",
        "cause":       "A root cause is: {name}. {desc}",
        "solution":    "A proposed solution is: {name}. {desc}",
        "abstraction": "The abstract pattern is: {name}. {desc}",
        "analogy":     "An analogous case: {name}. {desc}",
        "inspiration": "An inspiration: {name}. {desc}",
        "question":    "An open question: {name}. {desc}",
        "answer":      "An answer given: {name}. {desc}",
        "pro":         "A noted pro: {name}. {desc}",
        "con":         "A noted con: {name}. {desc}",
        "part":        "A required part: {name}. {desc}",
        "failure_mode": "A possible failure mode: {name}. {desc}",
    }

    parts = []
    for node in chain:
        tmpl = templates.get(node["ptype"], "{name}. {desc}")
        parts.append(tmpl.format(name=node["name"], desc=node.get("description", "")))
    return " ".join(parts)
```

```python
def _strip_long_form_directives(text: str) -> str:
    """
    In brief mode, physically remove the AIdeator directives that demand multi-paragraph,
    question-prefixed prose. Merely telling the model to "ignore" them was not enough —
    ~30% of descriptions still came back starting "Why is this important? ...".
    """
    for d in _PARAGRAPH_DIRECTIVES:
        text = text.replace(d, "")
    return " ".join(text.split(" "))
```

```python
def _content_instruction(
    action: str,
    post: dict,
    all_posts: dict,
    existing_by_ptype: dict | None = None,
    use_aideator: bool = False,
    brief_answers: bool = True,
) -> str:
    """Returns specific content instructions for the chosen action."""
    if use_aideator and existing_by_ptype is not None:
        from .aideator_templates import build_aideator_instruction
        ai = build_aideator_instruction(action, post, all_posts, existing_by_ptype)
        if ai is not None:
            return _strip_long_form_directives(ai) if brief_answers else ai

    ptype = post["ptype"]
    name = post["name"]
    desc = post.get("description", "")
```

```python
async def _elicit_skip_reason(self, post: dict, recommended_action: str, context_str: str) -> str:
    """
    Explicitly ASK the agent why it skipped a recommendation (Feature 2).

    Models reliably include a reason when they engage but tend to omit it when
    they skip, so we follow up with a focused one-shot question. The answer
    feeds back into the recommender as a negative-weight rule.
    """
    # IMPORTANT: do NOT reuse self.system_prompt here — it forces the agent to reply
    # in JSON, which would wrap the reason in {"action": ...}. Use a plain-prose
    # system instruction that still carries the persona voice.
    persona_system = (
        f"You are {self.name}, a {self.domain} expert participating in a structured "
        f"deliberation. {self.agent.persona_description} "
        "Answer in plain prose — full sentences, no JSON, no code fences."
    )
    prompt = (
        f"CONTEXT:\n{context_str}\n\n"
        f"You were just recommended to '{recommended_action}' on the "
        f"[{post['ptype'].upper()}] post \"{post['name']}\", and you chose to SKIP it.\n\n"
        "Explain WHY you skipped in ONE or TWO complete sentences — what made this a poor "
        "fit for you right now (off-topic for your expertise, already covered, lower "
        "priority than another branch, etc.). Write full sentences that finish the thought. "
        "Reply with the explanation only, no preamble, no JSON."
    )
    try:
        # Direct call with thinking DISABLED so the whole token budget goes to the
        # answer — otherwise Gemini's thinking step truncates short responses mid-sentence.
        resp = await _get_client().aio.models.generate_content(
            model=self.model_id,
            contents=prompt,
            config=genai_types.GenerateContentConfig(
                system_instruction=persona_system,
                max_output_tokens=260,
                temperature=0.5,
                thinking_config=genai_types.ThinkingConfig(thinking_budget=0),
            ),
        )
        text = (resp.text or "").strip().strip('"').strip()
        return _ensure_complete_sentence(text)[:400]
    except Exception:
        return ""
```

`backend/simulation/roles/specialist.py:23` — `sanitize_description`

```python
def sanitize_description(text: str) -> str:
    """
    Strip model formatting artifacts before persisting a post description.

    Observed in real runs: `</p>` tags and `**bold**` (verbose mode, ~24% of posts) and
    rhetorical-question openers like "Why is this important? ..." inherited from the
    AIdeator paragraph directives (~30% of posts in brief mode).
    """
    t = (text or "").strip()
    t = _MD_HTML.sub(" ", t)   # space, not empty: </p><p> must not weld words together
    t = _MD_MARKS.sub("", t)
    prev = None
    while prev != t:                      # openers can stack: "What is X? Why is Y? ..."
        prev = t
        t = _LEADING_Q.sub("", t).strip()
    return " ".join(t.split())
```

```python
def _summarize_tree(tree_state: dict) -> str:
    """Builds a compact text summary of the tree for LLM prompts."""
    all_posts = tree_state.get("all_posts", {})
    if not all_posts:
        return "Empty tree."

    from collections import Counter
    ptype_counts = Counter(p["ptype"] for p in all_posts.values())
    count_str = ", ".join(f"{v} {k}s" for k, v in sorted(ptype_counts.items()))

    # List posts by ptype
    lines = [f"Tree contains: {count_str}", ""]
    for ptype in ["mission", "stakeholder", "goal", "barrier", "cause", "solution",
                  "abstraction", "analogy", "inspiration", "question"]:
        posts_of_type = [p for p in all_posts.values() if p["ptype"] == ptype]
        if posts_of_type:
            lines.append(f"{ptype.upper()}S:")
            for p in posts_of_type[:8]:
                achiever_count = len([x for x in all_posts.values() if x.get("purpose_id") == p["id"]])
                lines.append(f"  [{p['id'][:8]}] {p['name']} ({achiever_count} children)")

    return "\n".join(lines)
```

```python
async def refine_persona_topics(db: Session, session_id: str) -> dict:
    """
    Skip-driven refinement (reduce the skip rate).

    For each agent, compare its topic headers against (a) the content it actually
    engaged with and (b) the reasons it gave for SKIPPING recommendations, using
    embedding similarity. Topics that look like what the agent engaged with get a
    confidence boost; topics that look like what it skipped get a penalty. Ranks are
    then re-ordered by the updated confidence WITHIN each sibling group (so sub-headers
    reorder among themselves and top-level headers among themselves).
    """
    from simulation.topic_tree import _embed_all, _cosine

    headers = (
        db.query(PersonaTopicHeader)
        .filter(PersonaTopicHeader.session_id == session_id)
        .order_by(PersonaTopicHeader.agent_id.asc(), PersonaTopicHeader.rank.asc())
        .all()
    )
    if not headers:
        return {"agents_refined": 0, "topics_updated": 0}

    headers_by_agent: dict = {}
    for h in headers:
        headers_by_agent.setdefault(h.agent_id, []).append(h)

    posts = (
        db.query(DeliberationPost)
        .filter(
            DeliberationPost.session_id == session_id,
            DeliberationPost.is_removed == False,
            DeliberationPost.author_agent_id != None,
        )
        .all()
    )
    posts_by_agent: dict = {}
    for p in posts:
        posts_by_agent.setdefault(p.author_agent_id, []).append(f"{p.name}. {p.description}")

    skip_logs = (
        db.query(RecommendationLog)
        .filter(
            RecommendationLog.session_id == session_id,
            RecommendationLog.outcome == "skipped",
            RecommendationLog.skip_reason.isnot(None),
        )
        .all()
    )
    skips_by_agent: dict = {}
    for l in skip_logs:
        txt = (l.skip_reason or "").strip()
        if txt and txt != "empty_content":
            skips_by_agent.setdefault(l.agent_id, []).append(txt)

    BOOST, PENALTY = 0.25, 0.25
    agents_refined = 0
    topics_updated = 0

    for agent_id, ahdrs in headers_by_agent.items():
        engaged_texts = posts_by_agent.get(agent_id, [])
        skip_texts = skips_by_agent.get(agent_id, [])
        if not engaged_texts and not skip_texts:
            continue

        header_vecs = await _embed_all([f"{h.label}. {h.rationale or ''}" for h in ahdrs])
        engaged_vecs = await _embed_all(engaged_texts) if engaged_texts else []
        skip_vecs = await _embed_all(skip_texts) if skip_texts else []

        for h, hv in zip(ahdrs, header_vecs):
            sim_engage = max((_cosine(hv, ev) for ev in engaged_vecs), default=0.0)
            sim_skip = max((_cosine(hv, sv) for sv in skip_vecs), default=0.0)
            base = h.confidence if h.confidence is not None else 0.5
            new_conf = base + BOOST * max(0.0, sim_engage) - PENALTY * max(0.0, sim_skip)
            h.confidence = round(max(0.05, min(0.99, new_conf)), 3)
            topics_updated += 1

        # Re-order ranks within each sibling group (same parent_id).
        siblings: dict = {}
        for h in ahdrs:
            siblings.setdefault(h.parent_id, []).append(h)
        for group in siblings.values():
            for rank, h in enumerate(sorted(group, key=lambda x: x.confidence or 0.0, reverse=True)):
                h.rank = rank
        agents_refined += 1

    db.commit()
    return {"agents_refined": agents_refined, "topics_updated": topics_updated}
```

async def split_persona_topic(db: Session, session_id: str, topic_id: str) -> dict:
    """
    Deepen one persona topic header by generating 2-3 finer SUB-HEADERS beneath it
    (LLM). The parent header is KEPT; the new sub-headers become its children and are
    given their own per-action scores. The parent's action_scores are recomputed as
    the mean of its children.
    """
    parent = (
        db.query(PersonaTopicHeader)
        .filter(PersonaTopicHeader.id == topic_id, PersonaTopicHeader.session_id == session_id)
        .first()
    )
    if not parent:
        return {"split": False, "reason": "topic not found"}

    agent = db.query(Agent).filter(Agent.id == parent.agent_id).first()
    persona = agent.persona_description if agent else ""

    prompt = (
        f"A deliberation participant has a persona topic of interest called "
        f'"{parent.label}" ({parent.rationale or ""}).\n\n'
        f"Participant persona: {persona}\n\n"
        "Split this single topic into 2 or 3 MORE SPECIFIC sub-topics that capture the "
        "distinct facets this participant most likely cares about. Each sub-topic gets a "
        "short header (2-5 words), a one-sentence rationale, and a confidence 0.0-1.0.\n\n"
        "Return ONLY valid JSON (no markdown):\n"
        '{"subtopics": [{"label": "...", "rationale": "...", "confidence": 0.8}, ...]}'
    )
    try:
        resp = await _get_client().aio.models.generate_content(
            model=TOPIC_MODEL,
            contents=prompt,
            config=genai_types.GenerateContentConfig(
                max_output_tokens=400, temperature=0.5,
                thinking_config=genai_types.ThinkingConfig(thinking_budget=0),
            ),
        )
        raw = re.sub(r"```(?:json)?", "", (resp.text or "").strip()).strip("`").strip()
        subs = json.loads(raw).get("subtopics", [])
    except Exception:
        subs = []

    subs = [s for s in subs if isinstance(s, dict) and str(s.get("label", "")).strip()][:3]
    if len(subs) < 2:
        return {"split": False, "reason": "could not split"}

    # Where to start ranking the new children (after any existing children).
    existing_children = (
        db.query(PersonaTopicHeader)
        .filter(PersonaTopicHeader.parent_id == parent.id)
        .count()
    )

    created = []
    new_rows: List[PersonaTopicHeader] = []
    for i, s in enumerate(subs):
        h = PersonaTopicHeader(
            session_id=session_id, agent_id=parent.agent_id,
            label=str(s["label"]).strip()[:80],
            rank=existing_children + i,
            rationale=str(s.get("rationale", ""))[:200],
            confidence=_clamp01(s.get("confidence", 0.5)),
            parent_id=parent.id,
        )
        db.add(h)
        new_rows.append(h)
        created.append(h.label)
    db.flush()

    # Score the new children's actions, then roll the parent up to their mean.
    await _score_actions_for_agent(persona, new_rows)
    all_children = (
        db.query(PersonaTopicHeader)
        .filter(PersonaTopicHeader.parent_id == parent.id)
        .all()
    )
    parent.action_scores = _mean_action_scores(
        [c.action_scores for c in all_children if c.action_scores]
    )
    db.commit()
    return {"split": True, "subtopics": created}
